        -- Rotate the main cursor.
        set({"n", "v"}, "<left>", mc.nextCursor)                                    "<left>"
        set({"n", "v"}, "<right>", mc.prevCursor)                                   "<right>"

        -- Delete the main cursor.
        set({"n", "v"}, "<leader>x", mc.deleteCursor)                                "<>x"     "x" (muchos con x de prefijo - trouble)

        set({"n", "v"}, "<leader>mx", mc.deleteCursor)

        -- Add and remove cursors with control + left click.
        set("n", "<c-leftmouse>", mc.handleMouse)                                    "<c-leftmouse>"

        -- Easy way to add and remove cursors using the main cursor.
        set({"n", "v"}, "<c-q>", mc.toggleCursor)                                   "<c-q>"

        -- Clone every cursor and disable the originals.
        set({"n", "v"}, "<leader><c-q>", mc.duplicateCursors)                       "<><c-q>"    ok

        set("n", "<esc>", function()                                                "<esc>"    ----
            if not mc.cursorsEnabled() then
                mc.enableCursors()
            elseif mc.hasCursors() then
                mc.clearCursors()
            else
                -- Default <esc> handler.
            end
        end)

        -- bring back cursors if you accidentally clear them
        set("n", "<leader>gv", mc.restoreCursors)                                    "<>gv"  "gv" (muchos con g de prefijo - git)

        -- Align cursor columns.
        set("n", "<leader>a", mc.alignCursors)                                       "<>a"  "a" ok

        -- Split visual selections by regex.
        set("v", "S", mc.splitCursors)                                                 

        -- Append/insert for each line of visual selections.
        set("v", "I", mc.insertVisual)                                             "I"
        set("v", "A", mc.appendVisual)                                             "A"

        -- match new cursors within visual selections by regex.
        set("v", "M", mc.matchCursors)                                            "M"

        -- Rotate visual selection contents.
        set("v", "<leader>t", mc.transposeCursors(1))                             "<>t"   "t"
            function() mc.transposeCursors(1) end)
        set("v", "<leader>T", mc.transposeCursors(-1))                            "<>T"   "T"
            function() mc.transposeCursors(-1) end) 

        -- Jumplist support
        set({"v", "n"}, "<c-i>", mc.jumpForward)                                 "<c-i>"
        set({"v", "n"}, "<c-o>", mc.jumpBackward)                                "<c-o>"
